### CHECK enum_membership
RATIONALE: Gold values for parameters that declare an enum are drawn from that enum. Outputs that invent a value outside the enum are wrong.
```python
def check(output, ctx):
    try:
        c = parse_call(output)
        if not c: return True
        fn = next((f for f in ctx['functions'] if f.get('name') == c.get('name')), None)
        if not fn: return True
        props = fn.get('parameters', {}).get('properties', {})
        for k, v in (c.get('arguments') or {}).items():
            p = props.get(k)
            if not isinstance(p, dict) or 'enum' not in p: continue
            en = [str(e).strip().lower() for e in p['enum']]
            vals = v if isinstance(v, list) else [v]
            for x in vals:
                if isinstance(x, (str, int, float)) and str(x).strip().lower() not in en:
                    try:
                        if any(float(x) == float(e) for e in en): continue
                    except Exception: pass
                    return False
        return True
    except Exception:
        return True
```

### CHECK email_values_in_request
RATIONALE: An email address passed as an argument must come from the conversation. A fabricated address is always wrong.
```python
def check(output, ctx):
    try:
        import re
        c = parse_call(output)
        if not c: return True
        req = ctx['request'].lower()
        def walk(v):
            if isinstance(v, str):
                for e in re.findall(r'[\w.+-]+@[\w-]+\.[\w.]+', v):
                    if e.lower().strip('.') not in req: return False
            elif isinstance(v, list):
                return all(walk(i) is not False for i in v)
            elif isinstance(v, dict):
                return all(walk(i) is not False for i in v.values())
            return True
        return walk(c.get('arguments') or {}) is not False
    except Exception:
        return True
```

### CHECK large_numbers_in_request
RATIONALE: Large precise numbers (e.g. loan amounts) come straight from the request. The check is skipped when multiplier words like "k" or "million" could explain a conversion.

### CHECK percent_value_consistent
RATIONALE: When the request states exactly one percentage, a percent-like argument must equal N or N/100. Other values are wrong.
```python
def check(output, ctx):
    try:
        import re
        c = parse_call(output)
        if not c: return True
        ps = set(re.findall(r'(\d+(?:\.\d+)?)\s*(?:%|percent)', ctx['request'].lower()))
        if len(ps) != 1: return True
        p = float(ps.pop())
        keys = [k for k in (c.get('arguments') or {}) if 'percent' in k.lower()]
        if len(keys) != 1: return True
        v = c['arguments'][keys[0]]
        if isinstance(v, (int, float)) and not isinstance(v, bool):
            return abs(v - p) < 1e-6 or abs(v - p / 100) < 1e-6
        return True
    except Exception:
        return True
```

### CHECK iso_date_not_invented_format
RATIONALE: If the request writes a date in non-ISO form and the schema does not ask for YYYY-MM-DD, gold keeps the user's form. Reformatting it to ISO is wrong.
```python
def check(output, ctx):
    try:
        import re
        c = parse_call(output)
        if not c: return True
        req = ctx['request']
        if re.search(r'\d{4}-\d{2}-\d{2}', req): return True
        fn = next((f for f in ctx['functions'] if f.get('name') == c.get('name')), None)
        if not fn: return True
        props = fn.get('parameters', {}).get('properties', {})
        for k, v in (c.get('arguments') or {}).items():
            if isinstance(v, str) and re.fullmatch(r'\s*\d{4}-\d{2}-\d{2}\s*', v):
                d = str(props.get(k, {}).get('description', '')).lower()
                if 'yyyy' in d or 'iso' in d or 'format' in d: continue
                if re.search(r'\b(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)[a-z]*\b', req.lower()) and re.search(r'\d', req):
                    return False
        return True
    except Exception:
        return True
```

### CHECK single_quoted_title_matches

### CHECK time_value_from_request
