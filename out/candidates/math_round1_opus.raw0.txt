### CHECK answer_is_finite
RATIONALE: Every gold answer is a finite real number, so a correct output never ends in inf or nan. This catches degenerate final answers that still parse as a number.
```python
def check(output, ctx):
    try:
        import math
        a = final_answer(output)
        if a is None:
            return True
        v = float(a.replace(',', '').replace('$', '').strip())
        return math.isfinite(v)
    except Exception:
        return True
```

### CHECK boxed_matches_final
RATIONALE: If the output states a \boxed{X} answer, a correct output's #### answer agrees with it. A mismatch shows a careless final line.
```python
def check(output, ctx):
    try:
        import re
        a = final_answer(output)
        if a is None:
            return True
        v = float(a.replace(',', '').replace('$', '').strip())
        bs = re.findall(r'\\boxed\{\s*\$?([-\d.,]+)', output)
        if not bs:
            return True
        b = float(bs[-1].replace(',', '').rstrip('.'))
        return abs(b - v) < 1e-6 * max(1, abs(v))
    except Exception:
        return True
```

### CHECK final_answer_appears_in_reasoning
RATIONALE: A correct step-by-step solution derives its answer, so the number appears before the #### line. A number that appears nowhere earlier is unsupported.
```python
def check(output, ctx):
    try:
        import re
        a = final_answer(output)
        if a is None or '####' not in output:
            return True
        v = float(a.replace(',', '').replace('$', '').strip())
        body = output[:output.rfind('####')]
        nums = re.findall(r'-?\d[\d,]*\.?\d*', body)
        if len(nums) < 3:
            return True
        for n in nums:
            try:
                if abs(float(n.replace(',', '').rstrip('.')) - v) < 1e-6 * max(1, abs(v)):
                    return True
            except Exception:
                pass
        return False
    except Exception:
        return True
```

### CHECK magnitude_vs_question_numbers
RATIONALE: The answer cannot exceed the product of all question numbers (each at least 2) times 1e3 with a 1e7 floor. This bound is generous enough for any correct answer and catches explosions.
```python
def check(output, ctx):
    try:
        import re
        nums = [float(n.replace(',', '')) for n in re.findall(r'\d[\d,]*\.?\d*', ctx['question']) if n.replace(',', '').replace('.', '').isdigit()]
        a = final_answer(output)
        if a is None:
            return True
        v = abs(float(a.replace(',', '').replace('$', '').strip()))
        p = 1.0
        for n in nums:
            p *= max(n, 2)
            if p > 1e15:
                return True
        return v <= max(p * 1000, 1e7)
    except Exception:
        return True
```

### CHECK final_line_single_number
RATIONALE: If the #### line contains two different numbers, at least one is not committed to, which is a sign of confusion. This only fires when the leading token is a plain number followed by another plain number.
```python
def check(output, ctx):
    try:
        import re
        m = re.findall(r'####(.*)', output)
        if not m:
            return True
        nums = re.findall(r'\d[\d,]*\.?\d*', m[-1])
        if len(nums) <= 1:
            return True
        vals = set(round(float(n.replace(',', '').rstrip('.')), 6) for n in nums)
        return len(vals) == 1 or not re.match(r'^\s*\$?[\d,.]+\s+(or|and)\s', m[-1])
    except Exception:
        return True
```
