### CHECK entities_are_strings
RATIONALE: The label lower-cases every entity, so a correct output can only contain strings. This catches outputs whose lists hold numbers, nested objects or nulls.
```python
def check(output, ctx):
    try:
        d = parse_json(output)
        if not isinstance(d, dict):
            return True
        for k in ["person", "location", "organization", "else"]:
            v = d.get(k, [])
            if isinstance(v, list):
                for e in v:
                    if not isinstance(e, str):
                        return False
        return True
    except Exception:
        return True
```

### CHECK token_aligned
RATIONALE: CoNLL entities are spans of whole tokens, so a correct entity must match a contiguous token sequence of the text (case-insensitive). This catches partial-token extractions.
```python
def check(output, ctx):
    try:
        d = parse_json(output)
        if not isinstance(d, dict):
            return True
        toks = [x.lower() for x in ctx['text'].split()]
        for k in ["person", "location", "organization", "else"]:
            v = d.get(k, [])
            if not isinstance(v, list):
                continue
            for e in v:
                if not isinstance(e, str):
                    continue
                et = e.strip().lower().split()
                if not et:
                    continue
                n = len(et)
                if not any(toks[i:i+n] == et for i in range(len(toks) - n + 1)):
                    return False
        return True
    except Exception:
        return True
```

### CHECK entity_capitalized_in_text
RATIONALE: Named entities in CoNLL contain at least one uppercase character where they occur in the text. The check accepts if any occurrence qualifies. It catches common-noun phrases like "wild card", "former president" and "world record".
```python
def check(output, ctx):
    try:
        d = parse_json(output)
        if not isinstance(d, dict):
            return True
        toks = ctx['text'].split()
        low = [x.lower() for x in toks]
        for k in ["person", "location", "organization", "else"]:
            v = d.get(k, [])
            if not isinstance(v, list):
                continue
            for e in v:
                if not isinstance(e, str):
                    continue
                et = e.strip().lower().split()
                n = len(et)
                if n == 0:
                    continue
                occ = [i for i in range(len(toks) - n + 1) if low[i:i+n] == et]
                if not occ:
                    continue
                if not any(any(c.isupper() for c in " ".join(toks[i:i+n])) for i in occ):
                    return False
        return True
    except Exception:
        return True
```

### CHECK count_le_occurrences
RATIONALE: Each gold entity is a distinct span of the text. So an entity string cannot appear across all fields more times than it occurs in the text. This catches the same span being duplicated or assigned to several types.
```python
def check(output, ctx):
    try:
        d = parse_json(output)
        if not isinstance(d, dict):
            return True
        low = [x.lower() for x in ctx['text'].split()]
        from collections import Counter
        c = Counter()
        for k in ["person", "location", "organization", "else"]:
            v = d.get(k, [])
            if isinstance(v, list):
                for e in v:
                    if isinstance(e, str):
                        c[" ".join(e.strip().lower().split())] += 1
        for s, cnt in c.items():
            et = s.split()
            n = len(et)
            if n == 0:
                continue
            occ = sum(1 for i in range(len(low) - n + 1) if low[i:i+n] == et)
            if occ and cnt > occ:
                return False
        return True
    except Exception:
        return True
```

### CHECK no_leading_lowercase_article
RATIONALE: CoNLL spans exclude a lowercase leading "the", "a" or "an" (for example "the Dodgers" is gold "Dodgers"). The check only rejects when every text occurrence starts with the lowercase article.
```python
def check(output, ctx):
    try:
        d = parse_json(output)
        if not isinstance(d, dict):
            return True
        toks = ctx['text'].split()
        low = [x.lower() for x in toks]
        for k in ["person", "location", "organization", "else"]:
            v = d.get(k, [])
            if not isinstance(v, list):
                continue
            for e in v:
                if not isinstance(e, str):
                    continue
                et = e.strip().lower().split()
                n = len(et)
                if n < 2 or et[0] not in ("the", "a", "an"):
                    continue
                occ = [i for i in range(len(toks) - n + 1) if low[i:i+n] == et]
                if occ and all(toks[i] in ("the", "a", "an") for i in occ):
                    return False
        return True
    except Exception:
        return True
```

### CHECK no_all_lowercase_multiword_else
RATIONALE: A multi-word "else" entry where every text occurrence is fully lowercase cannot be a named entity. This specifically catches phrases like "alien-led profit boost".
```python
def check(output, ctx):
    try:
        d = parse_json(output)
        if not isinstance(d, dict):
            return True
        toks = ctx['text'].split()
        low = [x.lower() for x in toks]

        if not isinstance(v, list):
            return True
        for e in v:
            if not isinstance(e, str):

    except Exception:
        return True
```

### CHECK no_number_only_token_edges
