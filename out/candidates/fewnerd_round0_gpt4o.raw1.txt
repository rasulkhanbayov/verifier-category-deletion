### CHECK_no_nonexistent_tokens
RATIONALE: Every correct entity must consist of tokens that appear in the text. This catches outputs where entities contain words not present in the input context.
```python
def check(output, ctx):
    try:
        data = parse_json(output)
        if data is None:
            return True
        text_tokens = set(ctx['text'].split())
        for field in data.values():
            for entity in field:
                if not set(entity.split()).issubset(text_tokens):
                    return False
        return True
    except:
        return True
```

### CHECK_no_other_field_with_named_entities
RATIONALE: Named entities (e.g., persons, locations) should not be misclassified as "other." This ensures proper categorization of named entities.
```python
def check(output, ctx):
    try:
        data = parse_json(output)
        if data is None:
            return True
        named_entity_fields = ["person", "location", "organization", "art", "building", "event", "product"]
        named_entities = set()
        for field in named_entity_fields:
            named_entities.update(data.get(field, []))
        for other in data.get("other", []):
            if other in named_entities:
                return False
        return True
    except:
        return True
```
