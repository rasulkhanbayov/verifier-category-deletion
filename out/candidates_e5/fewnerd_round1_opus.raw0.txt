### CHECK entity_is_token_span
RATIONALE: The label lower-cases and strips each entity and compares it to gold token spans, so every correct entity must equal a contiguous run of lower-cased text tokens joined by single spaces. This catches `It's`, `Dr. Martin Haskell`, `D & E procedure` and `Geysers Air Monitoring Programs (GAMP)`.
```python
def check(output, ctx):
    try:
        p = parse_json(output)
        if not isinstance(p, dict):
            return True
        text = ctx['text']
        if '  ' in text or any(c in text for c in '\t\n\r'):
            return True
        toks = text.lower().split(' ')
        spans = set()
        n = len(toks)
        for i in range(n):
            for j in range(i + 1, min(n, i + 40) + 1):
                spans.add(' '.join(toks[i:j]))
        for v in p.values():
            if not isinstance(v, list):
                continue
            for e in v:
                if not isinstance(e, str):
                    continue
                s = e.lower().strip()
                if len(s.split(' ')) > 40:
                    continue
                if s not in spans:
                    return False
        return True
    except Exception:
        return True
```

### CHECK global_count_le_occurrences
RATIONALE: Gold spans occupy distinct, non-overlapping positions, so a string repeated across all fields cannot appear more times than it occurs in the text. This catches `FIFA World Cup 2018` listed under both art and event.
```python
def check(output, ctx):
    try:
        p = parse_json(output)
        if not isinstance(p, dict):
            return True
        toks = ctx['text'].lower().split()
        from collections import Counter
        c = Counter()
        for v in p.values():
            if isinstance(v, list):
                for e in v:
                    if isinstance(e, str):
                        c[e.lower().strip()] += 1
        for s, k in c.items():
            st = s.split()
            m = len(st)
            if m == 0:
                continue
            occ = sum(1 for i in range(len(toks) - m + 1) if toks[i:i + m] == st)
            if occ == 0:
                continue
            if k > occ:
                return False
        return True
    except Exception:
        return True
```

### CHECK non_overlapping_assignment_exists
RATIONALE: All gold entity instances can be placed on distinct, non-overlapping token positions. A backtracking search rejects only when no such placement exists, and it accepts if any entity cannot be located or the search runs too long.
```python
def check(output, ctx):
    try:
        p = parse_json(output)
        if not isinstance(p, dict):
            return True
        toks = ctx['text'].lower().split()
        items = []
        for v in p.values():
            if isinstance(v, list):
                for e in v:
                    if isinstance(e, str):
                        items.append(e.lower().strip().split())
        cands = []
        for st in items:
            m = len(st)
            if m == 0:
                return True
            c = [i for i in range(len(toks) - m + 1) if toks[i:i + m] == st]
            if not c:
                return True
            cands.append((m, c))
        cands.sort(key=lambda x: len(x[1]))
        used = [False] * len(toks)
        steps = [0]

        def bt(k):
            steps[0] += 1
            if steps[0] > 20000:
                raise RuntimeError
            if k == len(cands):
                return True
            m, c = cands[k]
            for i in c:
                if not any(used[i:i + m]):
                    for j in range(i, i + m):
                        used[j] = True
                    if bt(k + 1):
                        return True
                    for j in range(i, i + m):
                        used[j] = False
            return False

        try:
            return bt(0)
        except RuntimeError:
            return True
    except Exception:
        return True
```

### CHECK entity_consecutive_tokens_adjacent
RATIONALE: Consecutive words of a gold span are adjacent tokens in the text. This catches entities that splice together non-adjacent words or drop words in between, such as `D & E procedure`.
```python
def check(output, ctx):
    try:
        p = parse_json(output)
        if not isinstance(p, dict):
            return True
        toks = ctx['text'].lower().split()
        pairs = set(zip(toks, toks[1:]))
        ts = set(toks)
        for v in p.values():
            if isinstance(v, list):
                for e in v:
                    if isinstance(e, str):
                        w = e.lower().split()
                        if not all(x in ts for x in w):
                            continue
                        for a, b in zip(w, w[1:]):
                            if (a, b) not in pairs:
                                return False
        return True
    except Exception:
        return True
```
